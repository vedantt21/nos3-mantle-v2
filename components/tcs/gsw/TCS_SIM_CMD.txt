# Implementation notes:
# These commands are OpenC3 backdoor controls for the NOS3 TCS simulator.  They
# do not go through the cFS TCS app or UART protocol; they publish command-bus
# strings directly to the simulator node.  Use them for test setup or fault
# injection, then use TCS target commands when verifying flight behavior.
#
# Mapping to simulator command_callback():
#   TCS_SIM_ENABLE      -> ENABLE
#   TCS_SIM_DISABLE     -> DISABLE and reset thermal state
#   TCS_SIM_MODE_AUTO   -> MODE=AUTO
#   TCS_SIM_MODE_MANUAL -> MODE=MANUAL
#   TCS_SIM_HEATER_ON   -> HEATER=ON, accepted only when simulator is MANUAL
#   TCS_SIM_HEATER_OFF  -> HEATER=OFF, accepted only when simulator is MANUAL
#   TCS_SIM_SET_STATUS  -> STATUS=<STATUS>, used to drive device-status errors
COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_ENABLE BIG_ENDIAN "Enable Tcs Sim"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"ENABLE"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_DISABLE BIG_ENDIAN "Disable Tcs Sim"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"DISABLE"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_MODE_AUTO BIG_ENDIAN "Set Tcs Sim Control Mode Auto"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"MODE=AUTO"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_MODE_MANUAL BIG_ENDIAN "Set Tcs Sim Control Mode Manual"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"MODE=MANUAL"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_HEATER_ON BIG_ENDIAN "Set Tcs Sim Heater On"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"HEATER=ON"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_HEATER_OFF BIG_ENDIAN "Set Tcs Sim Heater Off"
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"HEATER=OFF"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_SET_STATUS BIG_ENDIAN "Set Tcs Sim Status"
  APPEND_PARAMETER STATUS 32 UINT MIN_UINT32 MAX_UINT32 0
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"STATUS=<STATUS>"}'

COMMAND SIM_CMDBUS_BRIDGE TCS_SIM_SET_STATUS BIG_ENDIAN "Set Tcs Sim Status"
  APPEND_PARAMETER STATUS 32 UINT MIN_UINT32 MAX_UINT32 0
  APPEND_PARAMETER CMD_TEMPLATE 0 STRING '{"node":"tcs-sim-command","cmd":"STATUS=<STATUS>"}'
